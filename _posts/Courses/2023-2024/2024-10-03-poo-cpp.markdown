<p><strong>Class Definition</strong>:</p>
<pre><code class="language-cpp">class Rectangle {
private:
    double width;
    double height;

public:
    // Constructor
    Rectangle(double w, double h) : width(w), height(h) {}

    // Member functions
    double area() const {

<h4>2. Constructors and Destructors</h4>

<p>Special member functions for object lifecycle management:</p>

<pre><code class="language-cpp">class DynamicArray {
private:
    int* data;
    size_t size;

public:
    // Default constructor
    DynamicArray() : data(nullptr), size(0) {}

    // Parameterized constructor
    DynamicArray(size_t n) : size(n) {
        data = new int[size];
    }

    // Copy constructor
    DynamicArray(const DynamicArray&amp; other) : size(other.size) {
        data = new int[size];
        std::copy(other.data, other.data + size, data);
    }

    // Move constructor (C++11)
    DynamicArray(DynamicArray&amp;&amp; other) noexcept
        : data(other.data), size(other.size) {
        other.data = nullptr;
        other.size = 0;
    }

    // Destructor
    ~DynamicArray() {
        delete[] data;
    }

    // Copy assignment operator
    DynamicArray&amp; operator=(const DynamicArray&amp; other) {
        if (this != &amp;other) {
            delete[] data;
            size = other.size;
            data = new int[size];
            std::copy(other.data, other.data + size, data);
        }
        return *this;
    }
};</code></pre>

<p><strong>Rule of Three/Five</strong>: If you need to define a custom destructor, copy constructor, or copy assignment operator, you likely need all three (or five with move semantics).</p>

<p>Code reuse through class hierarchies:</p>

<pre><code class="language-cpp">class Shape {
protected:
    std::string color;

public:
    Shape(const std::string&amp; c) : color(c) {}
    virtual ~Shape() {}  // Virtual destructor

    virtual double area() const = 0;  // Pure virtual function
    virtual void draw() const {
        std::cout &lt;&lt; "Drawing a " &lt;&lt; color &lt;&lt; " shape\n";
    }
};

class Circle : public Shape {
private:
    double radius;

public:
    Circle(const std::string&amp; c, double r)
        : Shape(c), radius(r) {}

    double area() const override {
        return M_PI * radius * radius;
    }

    void draw() const override {
        std::cout &lt;&lt; "Drawing a " &lt;&lt; color &lt;&lt; " circle\n";
    }
};

class Rectangle : public Shape {
private:
    double width, height;

public:
    Rectangle(const std::string&amp; c, double w, double h)
        : Shape(c), width(w), height(h) {}

    double area() const override {
        return width * height;
    }
};</code></pre>

<p><strong>Inheritance Types</strong>:</p>
<ul>
<li><strong>Public Inheritance</strong>: Is-a relationship</li>
<li><strong>Protected Inheritance</strong>: Implementation inheritance</li>
<li><strong>Private Inheritance</strong>: Not commonly used</li>
</ul>

<h4>4. Polymorphism</h4>

<p>Dynamic dispatch through virtual functions:</p>

<pre><code class="language-cpp">void printArea(const Shape&amp; shape) {
    std::cout &lt;&lt; "Area: " &lt;&lt; shape.area() &lt;&lt; std::endl;
    shape.draw();
}

int main() {
    Circle circle("red", 5.0);
    Rectangle rect("blue", 4.0, 6.0);

    printArea(circle);     // Calls Circle::area()
    printArea(rect);       // Calls Rectangle::area()

    // Polymorphic container
    std::vector&lt;std::unique_ptr&lt;Shape&gt;&gt; shapes;
    shapes.push_back(std::make_unique&lt;Circle&gt;("green", 3.0));
    shapes.push_back(std::make_unique&lt;Rectangle&gt;("yellow", 2.0, 5.0));

    for (const auto&amp; shape : shapes) {
        shape-&gt;draw();
    }
}</code></pre>

<p><strong>Key Concepts</strong>:</p>
<ul>
<li><strong>Virtual Functions</strong>: Enable runtime polymorphism</li>
<li><strong>Virtual Destructor</strong>: Essential for base classes</li>
<li><strong>Pure Virtual Functions</strong>: Define abstract interfaces</li>
<li><strong>Override Keyword</strong>: Explicit override declaration (C++11)</li>
</ul>

<h4>5. Operator Overloading</h4>

<p>Defining custom behavior for operators:</p>

<pre><code class="language-cpp">class Complex {
private:
    double real, imag;

public:
    Complex(double r = 0, double i = 0) : real(r), imag(i) {}

    // Arithmetic operators
    Complex operator+(const Complex&amp; other) const {
        return Complex(real + other.real, imag + other.imag);
    }

    Complex operator*(const Complex&amp; other) const {
        return Complex(
            real * other.real - imag * other.imag,
            real * other.imag + imag * other.real
        );
    }

    // Comparison operator
    bool operator==(const Complex&amp; other) const {
        return real == other.real &amp;&amp; imag == other.imag;
    }

    // Stream operators (as friend functions)
    friend std::ostream&amp; operator&lt;&lt;(std::ostream&amp; os, const Complex&amp; c) {
        os &lt;&lt; c.real &lt;&lt; " + " &lt;&lt; c.imag &lt;&lt; "i";
        return os;
    }

    friend std::istream&amp; operator&gt;&gt;(std::istream&amp; is, Complex&amp; c) {
        is &gt;&gt; c.real &gt;&gt; c.imag;
        return is;
    }
};</code></pre>

<h4>6. Templates</h4>

<p>Generic programming for code reusability:</p>

<pre><code class="language-cpp">// Function template
template&lt;typename T&gt;
T max(T a, T b) {
    return (a &gt; b) ? a : b;
}

// Class template
template&lt;typename T&gt;
class Stack {
private:
    std::vector&lt;T&gt; elements;

public:
    void push(const T&amp; element) {
        elements.push_back(element);
    }

    T pop() {
        if (elements.empty()) {
            throw std::runtime_error("Stack is empty");
        }
        T value = elements.back();
        elements.pop_back();
        return value;
    }

    bool empty() const {
        return elements.empty();
    }

    size_t size() const {
        return elements.size();
    }
};

// Template specialization
template&lt;&gt;
class Stack&lt;bool&gt; {
    // Specialized implementation for bool
};</code></pre>

<p>Powerful collection of generic containers and algorithms:</p>

<p><strong>Containers</strong>:</p>
<pre><code class="language-cpp">// Sequential containers
std::vector&lt;int&gt; vec = {1, 2, 3, 4, 5};
std::list&lt;std::string&gt; lst;

// Associative containers

<p><strong>Algorithms</strong>:</p>
<pre><code class="language-cpp">#include &lt;algorithm&gt;

std::vector&lt;int&gt; numbers = {5, 2, 8, 1, 9};

// Sorting
std::sort(numbers.begin(), numbers.end());

// Finding
auto it = std::find(numbers.begin(), numbers.end(), 8);

// Transforming
std::transform(numbers.begin(), numbers.end(), numbers.begin(),
               [](int n) { return n * 2; });

// Accumulating
int sum = std::accumulate(numbers.begin(), numbers.end(), 0);</code></pre>

<h4>8. Smart Pointers (C++11)</h4>

<p>Automatic memory management:</p>

<pre><code class="language-cpp">#include &lt;memory&gt;

// Unique pointer (exclusive ownership)
std::unique_ptr&lt;int&gt; ptr1 = std::make_unique&lt;int&gt;(42);

// Shared pointer (shared ownership)
std::shared_ptr&lt;int&gt; ptr2 = std::make_shared&lt;int&gt;(100);
std::shared_ptr&lt;int&gt; ptr3 = ptr2;  // Reference count = 2

// Weak pointer (non-owning reference)
std::weak_ptr&lt;int&gt; weak = ptr2;

// Example with classes
class MyClass {
public:
    MyClass() { std::cout &lt;&lt; "Constructor\n"; }
    ~MyClass() { std::cout &lt;&lt; "Destructor\n"; }
    void doSomething() { std::cout &lt;&lt; "Doing something\n"; }
};

std::unique_ptr&lt;MyClass&gt; obj = std::make_unique&lt;MyClass&gt;();
obj-&gt;doSomething();
// Destructor called automatically when obj goes out of scope</code></pre>

<p>Robust error management:</p>

private:
    std::string message;

public:
    FileNotFoundException(const std::string&amp; filename)
        : message("File not found: " + filename) {}

    const char* what() const noexcept override {
        return message.c_str();
    }

void readFile(const std::string&amp; filename) {
    try {
        std::ifstream file(filename);
        if (!file.is_open()) {
            throw FileNotFoundException(filename);
        }
        // Read file...
    } catch (const FileNotFoundException&amp; e) {
        std::cerr &lt;&lt; "Error: " &lt;&lt; e.what() &lt;&lt; std::endl;

    }

<h4>10. Design Patterns</h4>

<p>Common software design solutions:</p>

<p><strong>Singleton Pattern</strong>:</p>
<pre><code class="language-cpp">class Database {
private:
    static Database* instance;
    Database() {}  // Private constructor

public:
    static Database* getInstance() {
        if (instance == nullptr) {
            instance = new Database();
        }
        return instance;
    }

    void connect() {
        std::cout &lt;&lt; "Connecting to database...\n";
    }
};</code></pre>

<p><strong>Factory Pattern</strong>:</p>
<pre><code class="language-cpp">class ShapeFactory {
public:
    static std::unique_ptr&lt;Shape&gt; createShape(const std::string&amp; type) {
        if (type == "circle") {
            return std::make_unique&lt;Circle&gt;("red", 5.0);
        } else if (type == "rectangle") {
            return std::make_unique&lt;Rectangle&gt;("blue", 4.0, 6.0);
        }
        return nullptr;
    }
};</code></pre>

    double perimeter() const {
        return 2 * (width + height);
    }

    // Accesseurs et mutateurs
    void setWidth(double w) { width = w; }
    double getWidth() const { return width; }
};</code></pre>

<li><strong>Constructeur/Destructeur</strong> : Initialisation et nettoyage des objets</li>
<li><strong>Fonctions membres</strong> : Methodes qui operent sur les donnees de l'objet</li>
<li><strong>Correction const</strong> : Utilisation de const pour les operations en lecture seule</li>
</ul>

<h4>2. Constructeurs et destructeurs</h4>

<p>Fonctions membres speciales pour la gestion du cycle de vie des objets :</p>

<pre><code class="language-cpp">class DynamicArray {
private:
    int* data;
    size_t size;

public:
    // Constructeur par defaut
    DynamicArray() : data(nullptr), size(0) {}

    // Constructeur parametre
    DynamicArray(size_t n) : size(n) {
        data = new int[size];
    }

    // Constructeur de copie
    DynamicArray(const DynamicArray&amp; other) : size(other.size) {
        data = new int[size];
        std::copy(other.data, other.data + size, data);
    }

    // Constructeur de deplacement (C++11)
    DynamicArray(DynamicArray&amp;&amp; other) noexcept
        : data(other.data), size(other.size) {
        other.data = nullptr;
        other.size = 0;
    }

    // Destructeur
    ~DynamicArray() {
        delete[] data;
    }

    // Operateur d'affectation par copie
    DynamicArray&amp; operator=(const DynamicArray&amp; other) {
        if (this != &amp;other) {
            delete[] data;
            size = other.size;
            data = new int[size];
            std::copy(other.data, other.data + size, data);
        }
        return *this;
    }
};</code></pre>

<p><strong>Regle des trois/cinq</strong> : Si vous devez definir un destructeur personnalise, un constructeur de copie ou un operateur d'affectation par copie, vous aurez probablement besoin des trois (ou cinq avec la semantique de deplacement).</p>

<p>Reutilisation du code a travers les hierarchies de classes :</p>

<pre><code class="language-cpp">class Shape {
protected:
    std::string color;

public:
    Shape(const std::string&amp; c) : color(c) {}
    virtual ~Shape() {}  // Destructeur virtuel

    virtual double area() const = 0;  // Fonction virtuelle pure
    virtual void draw() const {
        std::cout &lt;&lt; "Drawing a " &lt;&lt; color &lt;&lt; " shape\n";
    }
};

class Circle : public Shape {
private:
    double radius;

public:
    Circle(const std::string&amp; c, double r)
        : Shape(c), radius(r) {}

    double area() const override {
        return M_PI * radius * radius;
    }

    void draw() const override {
        std::cout &lt;&lt; "Drawing a " &lt;&lt; color &lt;&lt; " circle\n";
    }
};

class Rectangle : public Shape {
private:
    double width, height;

public:
    Rectangle(const std::string&amp; c, double w, double h)
        : Shape(c), width(w), height(h) {}

    double area() const override {
        return width * height;
    }
};</code></pre>

<p><strong>Types d'heritage</strong> :</p>
<ul>
<li><strong>Heritage public</strong> : Relation est-un</li>
<li><strong>Heritage protege</strong> : Heritage d'implementation</li>
<li><strong>Heritage prive</strong> : Rarement utilise</li>
</ul>

<h4>4. Polymorphisme</h4>

<p>Dispatch dynamique a travers les fonctions virtuelles :</p>

<pre><code class="language-cpp">void printArea(const Shape&amp; shape) {
    std::cout &lt;&lt; "Area: " &lt;&lt; shape.area() &lt;&lt; std::endl;
    shape.draw();
}

int main() {
    Circle circle("red", 5.0);
    Rectangle rect("blue", 4.0, 6.0);

    printArea(circle);     // Appelle Circle::area()
    printArea(rect);       // Appelle Rectangle::area()

    // Conteneur polymorphe
    std::vector&lt;std::unique_ptr&lt;Shape&gt;&gt; shapes;
    shapes.push_back(std::make_unique&lt;Circle&gt;("green", 3.0));
    shapes.push_back(std::make_unique&lt;Rectangle&gt;("yellow", 2.0, 5.0));

    for (const auto&amp; shape : shapes) {
        shape-&gt;draw();
    }
}</code></pre>

<p><strong>Concepts cles</strong> :</p>
<ul>
<li><strong>Fonctions virtuelles</strong> : Permettent le polymorphisme a l'execution</li>
<li><strong>Destructeur virtuel</strong> : Essentiel pour les classes de base</li>
<li><strong>Fonctions virtuelles pures</strong> : Definissent des interfaces abstraites</li>
<li><strong>Mot-cle override</strong> : Declaration explicite de redefinition (C++11)</li>
</ul>

<h4>5. Surcharge d'operateurs</h4>

<p>Definition de comportements personnalises pour les operateurs :</p>

<pre><code class="language-cpp">class Complex {
private:
    double real, imag;

public:
    Complex(double r = 0, double i = 0) : real(r), imag(i) {}

    // Operateurs arithmetiques
    Complex operator+(const Complex&amp; other) const {
        return Complex(real + other.real, imag + other.imag);
    }

    Complex operator*(const Complex&amp; other) const {
        return Complex(
            real * other.real - imag * other.imag,
            real * other.imag + imag * other.real
        );
    }

    // Operateur de comparaison
    bool operator==(const Complex&amp; other) const {
        return real == other.real &amp;&amp; imag == other.imag;
    }

    // Operateurs de flux (comme fonctions amies)
    friend std::ostream&amp; operator&lt;&lt;(std::ostream&amp; os, const Complex&amp; c) {
        os &lt;&lt; c.real &lt;&lt; " + " &lt;&lt; c.imag &lt;&lt; "i";
        return os;
    }

    friend std::istream&amp; operator&gt;&gt;(std::istream&amp; is, Complex&amp; c) {
        is &gt;&gt; c.real &gt;&gt; c.imag;
        return is;
    }
};</code></pre>

<h4>6. Templates</h4>

<pre><code class="language-cpp">// Template de fonction
template&lt;typename T&gt;
T max(T a, T b) {
    return (a &gt; b) ? a : b;
}

// Template de classe
template&lt;typename T&gt;
class Stack {
private:
    std::vector&lt;T&gt; elements;

public:
    void push(const T&amp; element) {
        elements.push_back(element);
    }

    T pop() {
        if (elements.empty()) {
            throw std::runtime_error("Stack is empty");
        }
        T value = elements.back();
        elements.pop_back();
        return value;
    }

    bool empty() const {
        return elements.empty();
    }

    size_t size() const {
        return elements.size();
    }
};

// Specialisation de template
template&lt;&gt;
class Stack&lt;bool&gt; {
    // Implementation specialisee pour bool
};</code></pre>

<h4>9. Gestion des exceptions</h4>

<pre><code class="language-cpp">class FileNotFoundException : public std::exception {
private:
    std::string message;

public:
    FileNotFoundException(const std::string&amp; filename)
        : message("Fichier non trouve : " + filename) {}

    const char* what() const noexcept override {
        return message.c_str();
    }
};

void readFile(const std::string&amp; filename) {
    try {
        std::ifstream file(filename);
        if (!file.is_open()) {
            throw FileNotFoundException(filename);
        }
        // Lire le fichier...
    } catch (const FileNotFoundException&amp; e) {
        std::cerr &lt;&lt; "Erreur : " &lt;&lt; e.what() &lt;&lt; std::endl;
        throw;  // Re-lancer
    } catch (const std::exception&amp; e) {
        std::cerr &lt;&lt; "Erreur inattendue : " &lt;&lt; e.what() &lt;&lt; std::endl;
    }
}</code></pre>
